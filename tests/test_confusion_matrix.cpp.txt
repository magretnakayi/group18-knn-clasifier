#include <iostream>
#include <cassert>
#include "ConfusionMatrix.hpp"
using namespace std;
int main() {
    cout << "Running Confusion Matrix Tests..." << endl;
    int num_samples = 4;
    int num_classes = 2;
    int* true_labels = new int[num_samples]{0, 0, 1, 1};
    int* pred_labels = new int[num_samples]{0, 1, 1, 1};
    ConfusionMatrix cm(true_labels, pred_labels, num_samples, num_classes);
    double accuracy = cm.getAccuracy();
    assert(accuracy == 0.75); // 3 out of 4 correct
    cout << "[PASS] Confusion Matrix Accuracy Verification (75%) Passed!" << endl;
    delete[] true_labels;
    delete[] pred_labels;
    return 0;
}
